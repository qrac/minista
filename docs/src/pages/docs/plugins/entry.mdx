# pluginEntry

ルートパスのCSS・JavaScript・画像をビルドプロセスに乗せるプラグイン。

## Table of Contents

contents list

## How To Use

```js title="./vite.config.js"
import { pluginSsg, pluginEntry } from "minista"

export default {
  plugins: [pluginSsg(), pluginEntry()],
}
```

```jsx title="./src/pages/index.jsx"
import { Head } from "minista/head"

export default function () {
  return (
    <>
      <Head>
        <link rel="stylesheet" href="/src/assets/style.css" />
        <script type="module" src="/src/assets/script.ts" />
      </Head>
      <div>
        <img src="/src/assets/image.png" width={76} height={76} />
      </div>
    </>
  )
}
```

```js
// ❌ Unsupported paths.
<link rel="stylesheet" href="src/assets/style.css" /> // Relative path
<link rel="stylesheet" href="../src/assets/style.css" /> // Relative path

// ❌ Imports are not supported
import "/src/assets/style.css" // Used for pluginBundle
import "/src/assets/script.ts" // Removed from HTML
```
